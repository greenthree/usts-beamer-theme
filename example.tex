\begin{frame}{数学公式}
  \begin{block}{递推关系}
    当 $n \ge m$ 时，有
    \[
      E(n) = 1 + \frac{1}{m}\sum_{i=1}^{m}E(n-i).
    \]
  \end{block}
  \begin{itemize}
    \item 行内公式：$E=mc^2$。
    \item 多行对齐公式：
      \[
        \begin{aligned}
          i - \operatorname{pre}(i)
            &\le \operatorname{pre}(n)-\operatorname{pre}(i),\\
          i &\le \operatorname{pre}(n).
        \end{aligned}
      \]
  \end{itemize}
\end{frame}

\begin{frame}{定义、定理与证明}
  \begin{definition}
    若数列满足 $a_{n+1}=q a_n$，则称其为等比数列。
  \end{definition}
  \begin{theorem}
    \upshape % 中文字体通常没有独立的斜体字形
    公比为 $q$ 的等比数列满足 $a_n=a_1q^{n-1}$。
  \end{theorem}
  \begin{proof}
    由递推关系连续代入即可得到结论。
  \end{proof}
\end{frame}
